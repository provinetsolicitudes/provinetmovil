<script>
(function() {
    var cedulaInput = document.getElementById('cedula');
    if (!cedulaInput) return;

    var realValue = '';
    var displayValue = [];  // array of {char, masked}
    var maskTimer = null;

    function updateDisplay() {
        var shown = displayValue.map(function(d) {
            return d.masked ? '*' : d.char;
        }).join('');
        cedulaInput.value = shown;
        var hidden = document.getElementById('cedula_real');
        if (hidden) hidden.value = realValue;
        cedulaInput.dispatchEvent(new Event('input'));
    }

    function maskLastVisible() {
        // Find last unmasked character and mask it
        for (var i = displayValue.length - 1; i >= 0; i--) {
            if (!displayValue[i].masked) {
                displayValue[i].masked = true;
                break;
            }
        }
        updateDisplay();
    }

    cedulaInput.addEventListener('keydown', function(e) {
        var key = e.key;
        var maxLen = parseInt(cedulaInput.getAttribute('maxlength') || 10);

        if (key === 'Backspace') {
            if (maskTimer) clearTimeout(maskTimer);
            maskTimer = null;
            realValue = realValue.slice(0, -1);
            displayValue = displayValue.slice(0, -1);
            updateDisplay();
            e.preventDefault();
        } else if (key === 'Delete') {
            if (maskTimer) clearTimeout(maskTimer);
            maskTimer = null;
            realValue = '';
            displayValue = [];
            updateDisplay();
            e.preventDefault();
        } else if (/^[0-9]$/.test(key) && realValue.length < maxLen) {
            e.preventDefault();

            // Immediately mask previous visible char
            if (maskTimer) {
                clearTimeout(maskTimer);
                maskTimer = null;
                maskLastVisible();
            }

            // Add new char as visible
            realValue += key;
            displayValue.push({ char: key, masked: false });
            updateDisplay();

            // Schedule masking after 1 second
            maskTimer = setTimeout(function() {
                maskLastVisible();
                maskTimer = null;
            }, 1000);
        }
        // Let Tab, Enter, arrows work normally (no preventDefault)
    });
})();
</script>

<script>
(function() {
    var pwdInput = document.getElementById('password');
    if (!pwdInput) return;

    // Create eye toggle button
    var eyeBtn = document.createElement('button');
    eyeBtn.type = 'button';
    eyeBtn.id = 'eye-toggle';
    eyeBtn.style.cssText = [
        'position:absolute',
        'top:62%',
        'right:12px',
        'transform:translateY(-50%)',
        'background:none',
        'border:none',
        'cursor:pointer',
        'padding:0',
        'display:none',
        'z-index:10',
        'color:#1464a5'
    ].join(';');

    var eyeOpen = '<svg xmlns="http://www.w3.org/2000/svg" width="22" height="22" viewBox="0 0 24 24" fill="none" stroke="#1464a5" stroke-width="2" stroke-linecap="round" stroke-linejoin="round"><path d="M1 12s4-8 11-8 11 8 11 8-4 8-11 8-11-8-11-8z"/><circle cx="12" cy="12" r="3"/></svg>';
    var eyeClosed = '<svg xmlns="http://www.w3.org/2000/svg" width="22" height="22" viewBox="0 0 24 24" fill="none" stroke="#1464a5" stroke-width="2" stroke-linecap="round" stroke-linejoin="round"><path d="M17.94 17.94A10.07 10.07 0 0 1 12 20c-7 0-11-8-11-8a18.45 18.45 0 0 1 5.06-5.94"/><path d="M9.9 4.24A9.12 9.12 0 0 1 12 4c7 0 11 8 11 8a18.5 18.5 0 0 1-2.16 3.19"/><line x1="1" y1="1" x2="23" y2="23"/></svg>';

    eyeBtn.innerHTML = eyeOpen;
    var visible = false;

    eyeBtn.addEventListener('click', function() {
        visible = !visible;
        pwdInput.type = visible ? 'text' : 'password';
        eyeBtn.innerHTML = visible ? eyeClosed : eyeOpen;
    });

    // Show eye icon when user starts typing, hide when empty
    pwdInput.addEventListener('input', function() {
        eyeBtn.style.display = pwdInput.value.length > 0 ? 'block' : 'none';
        if (pwdInput.value.length === 0) {
            visible = false;
            pwdInput.type = 'password';
            eyeBtn.innerHTML = eyeOpen;
        }
    });

    // Insert after password input
    pwdInput.parentNode.style.position = 'relative';
    pwdInput.parentNode.appendChild(eyeBtn);
})();
</script>
